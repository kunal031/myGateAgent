# Dependencies for the deployed functions.
#
# Separate from requirements.txt because that file also carries the
# ingestion toolchain -- gdown, python-docx, python-pptx -- which only
# the local CLIs use. Embedding a book takes hours against Lambda's
# 15-minute ceiling, so ingestion never runs there and shipping its
# dependencies would spend a third of the 250 MB unzipped limit on code
# that cannot execute.
#
# pymupdf stays. It is imported lazily inside import_paper_pdf, so a
# naive import scan reports it unused -- it is not. The worker needs it
# to read an uploaded question paper.
#
# boto3 is provided by the Lambda runtime and deliberately omitted here;
# it stays pinned in requirements.txt for local runs.
#
# This file is the API's dependencies. The worker additionally needs the
# document toolchain (requirements-lambda-worker.txt), which is 92 MB of
# pymupdf, lxml and Pillow that no API route touches. Shipping one set to
# both put the package at 230 MB against a 250 MB limit -- close enough
# that a single dependency update would have broken the deploy.

# ---- Web ----
fastapi==0.142.2
pydantic==2.13.5
pydantic-settings==2.15.0
python-multipart==0.0.32
mangum==0.19.0

# ---- Auth ----
PyJWT==2.12.1
cryptography==50.0.2
bcrypt==5.0.0
email-validator==2.3.0

# ---- Database ----
SQLAlchemy==2.1.3
# 3.2.13, not 3.3.x: psycopg-binary 3.3.x publishes no aarch64
# Linux wheel, so pip backtracks through every other pin looking
# for a combination that works and fails on numpy instead -- which
# is a misleading error for a psycopg problem. Local
# requirements.txt keeps 3.3.6; this pin is only for Lambda.
psycopg[binary]==3.2.13
pgvector==0.5.1

# ---- AI ----
google-genai==2.29.0
# 2.2.6, not 2.4.x: from 2.3 the aarch64 wheels require
# manylinux_2_28, and SAM's build image is manylinux2014 (glibc
# 2.17). pip then tries to compile numpy from source inside the
# container and fails on 'NumPy requires GCC >= 9.3'. The local
# requirements.txt is unaffected -- this pin is only for the
# Lambda package.
numpy==2.2.6
tenacity==9.1.4

httpx==0.28.1
# The API imports services/import_paper/fetch.py at module level -- it is
# a route handler's dependency, not a lazy one -- and that module parses
# a fetched question page with BeautifulSoup(html, "lxml"). Removing
# these broke every request with ModuleNotFoundError: No module named
# 'bs4'. Together they are 12.6 MB against pymupdf's 54 MB, which stays
# out because its only caller is a background job that runs in the
# worker.
beautifulsoup4==4.15.0
lxml==6.1.3
python-dotenv==1.2.4
